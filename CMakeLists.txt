cmake_minimum_required(VERSION 3.15)
project(cjs_module_lexer VERSION 2.3.0 LANGUAGES C)

add_library(cjs_module_lexer STATIC src/lexer.c src/native.c)
add_library(cjs_module_lexer::cjs_module_lexer ALIAS cjs_module_lexer)
target_compile_features(cjs_module_lexer PUBLIC c_std_11)
target_include_directories(cjs_module_lexer PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
if(MSVC)
  target_compile_options(cjs_module_lexer PRIVATE /W4)
else()
  target_compile_options(cjs_module_lexer PRIVATE -Wall -Wextra -Wno-parentheses)
endif()

include(GNUInstallDirs)
install(TARGETS cjs_module_lexer EXPORT cjs_module_lexer_targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/cjs-module-lexer.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT cjs_module_lexer_targets FILE cjs_module_lexer-config.cmake
  NAMESPACE cjs_module_lexer:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/cjs_module_lexer)

include(CTest)
if(BUILD_TESTING)
  enable_language(CXX)
  find_package(Threads REQUIRED)
  add_executable(native_test test/native/test.c)
  target_link_libraries(native_test PRIVATE cjs_module_lexer)
  add_test(NAME native COMMAND native_test)
  add_executable(native_parse test/native/parse.c)
  target_link_libraries(native_parse PRIVATE cjs_module_lexer)
  add_executable(native_threads test/native/threads.cc)
  target_compile_features(native_threads PRIVATE cxx_std_11)
  target_link_libraries(native_threads PRIVATE cjs_module_lexer Threads::Threads)
  add_test(NAME native_threads COMMAND native_threads)
  add_library(native_failure OBJECT src/lexer.c src/native.c)
  target_include_directories(native_failure PRIVATE include)
  target_compile_features(native_failure PRIVATE c_std_11)
  target_compile_definitions(native_failure PRIVATE malloc=cjs_test_malloc realloc=cjs_test_realloc free=cjs_test_free)
  add_executable(native_failures test/native/failures.c $<TARGET_OBJECTS:native_failure>)
  target_include_directories(native_failures PRIVATE include)
  add_test(NAME native_failures COMMAND native_failures)
  foreach(target native_test native_threads native_failures)
    if(MSVC)
      target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
  endforeach()
endif()
