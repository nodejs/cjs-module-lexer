cmake_minimum_required(VERSION 3.15)
project(consumer LANGUAGES C CXX)
find_package(cjs_module_lexer CONFIG REQUIRED)
add_executable(consumer_c main.c)
add_executable(consumer_cpp main.cc)
target_link_libraries(consumer_c PRIVATE cjs_module_lexer::cjs_module_lexer)
target_link_libraries(consumer_cpp PRIVATE cjs_module_lexer::cjs_module_lexer)
enable_testing()
add_test(NAME consumer_c COMMAND consumer_c)
add_test(NAME consumer_cpp COMMAND consumer_cpp)
